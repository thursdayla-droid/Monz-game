import { it } from 'vitest';
import { instinctFor } from '../src/core/autobattle/engine';
import { ANIME } from '../src/core/battle3d/animeConfig';
import { animeFinished, newAnimeBattle, stepAnime, type AnimeEntrant } from '../src/core/battle3d/anime';
import { growTechs } from '../src/core/battle3d/techLearn';
import type { StatBlock } from '../src/core/stats';
const S = (o: Partial<StatBlock>): StatBlock => ({ hp: 30, power: 36, tough: 32, speed: 12, tech: 16, ...o });
const B = { P: S({ power: 70 }), T: S({ tough: 70 }), Sp: S({ speed: 50 }), Te: S({ tech: 50 }) };
const ent = (level: number, st: StatBlock): AnimeEntrant => { const g = growTechs('bunk', level, st, undefined, instinctFor('bunk', st)); return { name: 'B', species: 'bunk', stats: st, instinct: instinctFor('bunk', st), proficiency: { attack: 2, defense: 2, evasion: 2, counter: 2, spacing: 2 }, techs: g.state.equipped }; };
it('bal', { timeout: 120000 }, () => {
  for (const lv of [10, 25, 40, 60]) {
    const keys = Object.keys(B) as (keyof typeof B)[];
    const row: string[] = [];
    let tsum = 0, n = 0;
    for (const a of keys) {
      const cells: string[] = [];
      for (const c of keys) {
        let w = 0; const N = 16;
        for (let s = 1; s <= N; s++) for (const swap of [false, true]) {
          const A = ent(lv, B[a]); const C = ent(lv, B[c]);
          const b = newAnimeBattle(swap ? C : A, swap ? A : C, s);
          while (!animeFinished(b) && b.t < 200) { b.events = []; stepAnime(b, ANIME.dt); }
          tsum += b.koAt ?? b.t; n++;
          if (b.winner === (swap ? 1 : 0)) w++;
        }
        cells.push(String(Math.round(w / (2 * N) * 100)).padStart(3));
      }
      row.push(`${a.padEnd(3)} ${cells.join(' ')}  ${ent(lv, B[a]).techs!.join('/')}`);
    }
    console.log(`Lv${lv} avg ${(tsum / n).toFixed(1)}s\n` + row.join('\n'));
  }
});
